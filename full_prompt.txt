**`src/lib/auth.ts`:**
```typescript
import {
  signInWithEmailAndPassword,
  onAuthStateChanged,
  type User,
} from 'firebase/auth';
import { doc, getDoc } from 'firebase/firestore';
import { auth, db } from './firebase';
import { useAuthStore } from '../store/auth.store';
import type { UserProfile } from '../types';

export async function loginUser(email: string, password: string): Promise<UserProfile> {
  const credential = await signInWithEmailAndPassword(auth, email, password);
  const profile = await fetchUserProfile(credential.user.uid);
  return profile;
}

export async function fetchUserProfile(uid: string): Promise<UserProfile> {
  const snap = await getDoc(doc(db, 'users', uid));
  if (!snap.exists()) {
    await auth.signOut();
    throw new Error('Account not configured. Contact your administrator.');
  }
  return { id: snap.id, ...snap.data() } as UserProfile;
}

// Call this once at app root â€” listens for auth state changes
export function initAuthListener(): () => void {
  return onAuthStateChanged(auth, async (firebaseUser: User | null) => {
    const { setUser, setLoading } = useAuthStore.getState();
    if (firebaseUser) {
      try {
        const profile = await fetchUserProfile(firebaseUser.uid);
        setUser(profile);
      } catch {
        setUser(null);
      }
    } else {
      setUser(null);
    }
    setLoading(false);
  });
}

// Atomic patient ID generation â€” never use count()
export async function generatePatientRefId(): Promise<string> {
  const seqRef = doc(db, 'sequences', 'patients');
  let newId = '';
  await runTransaction(db, async (tx) => {
    const snap = await tx.get(seqRef);
    const current = snap.exists() ? (snap.data().current as number) : 0;
    const prefix = snap.exists() ? (snap.data().prefix as string) : 'ASK';
    const next = current + 1;
    tx.set(seqRef, { current: next, prefix }, { merge: true });
    newId = `${prefix}-${String(next).padStart(4, '0')}`;
  });
  return newId;
}

export async function createPatient(
  data: Omit<Patient, 'id' | 'createdAt' | 'updatedAt' | 'patientRefId' | 'visitCount'>,
  createdBy: string,
): Promise<Patient> {
  const patientRefId = await generatePatientRefId();
  const ref = await addDoc(collection(db, 'patients'), {
    ...data,
    patientRefId,
    visitCount: 0,
    isActive: true,
    allergies: data.allergies ?? [],
    createdBy,
    createdAt: serverTimestamp(),
    updatedAt: serverTimestamp(),
  });
  const snap = await getDoc(ref);
  return { id: snap.id, ...snap.data() } as Patient;
}

export async function getPatient(patientId: string): Promise<Patient> {
  const snap = await getDoc(doc(db, 'patients', patientId));
  if (!snap.exists()) throw new Error('Patient not found');
  return { id: snap.id, ...snap.data() } as Patient;
}

  await updateDoc(doc(db, 'patients', patientId), {

export async function listPatients(opts: {
  search?: string;
  status?: string;
  pageSize?: number;
  lastDoc?: DocumentSnapshot;
}): Promise<{ patients: Patient[]; lastDoc: DocumentSnapshot | null }> {
  let q = query(
    collection(db, 'patients'),
    where('deletedAt', '==', null),
    orderBy('name'),
    limit(opts.pageSize ?? 25),
  );
  if (opts.lastDoc) q = query(q, startAfter(opts.lastDoc));
  const snap = await getDocs(q);
  const patients = snap.docs.map(d => ({ id: d.id, ...d.data() } as Patient));
  // Filter by search client-side (Firestore prefix search has limitations)
  const filtered = opts.search
    ? patients.filter(p =>
        p.name.toLowerCase().includes(opts.search!.toLowerCase()) ||
        p.mobile.includes(opts.search!) ||
        p.patientRefId.toLowerCase().includes(opts.search!.toLowerCase()),
      )
    : patients;
  return {
    patients: filtered,
    lastDoc: snap.docs[snap.docs.length - 1] ?? null,
  };
}

export async function searchPatients(q: string): Promise<Patient[]> {
  // Prefix search by name
  const snap = await getDocs(
    query(
      collection(db, 'patients'),
      where('deletedAt', '==', null),
      where('name', '>=', q),
      where('name', '<=', q + '\uf8ff'),
      limit(8),
    ),
  );
  return snap.docs.map(d => ({ id: d.id, ...d.data() } as Patient));
}

export async function uploadPatientPhoto(
  patientId: string,
  file: File,
): Promise<string> {
  const ext = file.name.split('.').pop();
  const storageRef = ref(storage, `patients/${patientId}/photo.${ext}`);
  await uploadBytes(storageRef, file);
  const url = await getDownloadURL(storageRef);
  await updateDoc(doc(db, 'patients', patientId), { photoUrl: url });
  return url;
}

// Visit number generation â€” MUST be inside a transaction
export async function createVisit(
  patientId: string,
  doctorId: string,
  doctorName: string,
): Promise<Visit> {
  const visitsRef = collection(db, 'patients', patientId, 'visits');
  let newVisit!: Visit;

  await runTransaction(db, async (tx) => {
    // Get highest visitNumber for this patient
    const q = query(visitsRef, orderBy('visitNumber', 'desc'), limit(1));
    const snap = await getDocs(q);
    const lastNum = snap.empty ? 0 : (snap.docs[0].data().visitNumber as number);

    const visitRef = doc(visitsRef);
    const visitData = {
      visitNumber: lastNum + 1,
      patientId,
      doctorId,
      doctorName,
      date: serverTimestamp(),
      isLocked: false,
      isDraft: true,
      paymentStatus: 'PENDING' as const,
      createdAt: serverTimestamp(),
      updatedAt: serverTimestamp(),
    };
    tx.set(visitRef, visitData);
    newVisit = { id: visitRef.id, ...visitData } as unknown as Visit;
  });

  return newVisit;
}

export async function getVisit(
  patientId: string,
  visitId: string,
): Promise<Visit> {
  const snap = await getDoc(doc(db, 'patients', patientId, 'visits', visitId));
  if (!snap.exists()) throw new Error('Visit not found');

  // Auto-lock check
  const visit = { id: snap.id, ...snap.data() } as Visit;
  if (!visit.isLocked && visit.createdAt) {
    const hoursSince = (Date.now() - visit.createdAt.toMillis()) / 3_600_000;
    if (hoursSince >= 24) {
      await updateDoc(snap.ref, { isLocked: true });
      visit.isLocked = true;
    }
  }
  return visit;
}

export async function updateVisit(
  patientId: string,
  visitId: string,
  data: Partial<Visit>,
): Promise<void> {
  await updateDoc(
    doc(db, 'patients', patientId, 'visits', visitId),
    { ...data, updatedAt: serverTimestamp() },
  );
}

export async function listVisits(patientId: string): Promise<Visit[]> {
  const snap = await getDocs(
    query(
      collection(db, 'patients', patientId, 'visits'),
      where('deletedAt', '==', null),
      orderBy('visitNumber', 'desc'),
    ),
  );
  return snap.docs.map(d => ({ id: d.id, ...d.data() } as Visit));
}

export async function saveVisit(
  patientId: string,
  visitId: string,
  data: Partial<Visit>,
): Promise<void> {
  // Mark as no longer draft
  await updateVisit(patientId, visitId, { ...data, isDraft: false });

  // Update patient denormalised fields
  await updateDoc(doc(db, 'patients', patientId), {
    lastVisitDate: new Date().toISOString().split('T')[0],
    nextFollowUpDate: data.followUpDate ?? null,
    updatedAt: serverTimestamp(),
  });

  // Auto-create or update invoice if fee exists
  if (data.totalFee && data.totalFee > 0) {
    await upsertInvoiceForVisit(patientId, visitId, data);
  }
}

export async function repeatLastPrescription(
  patientId: string,
  targetVisitId: string,
): Promise<void> {
  // Find all visits ordered by number desc
  const visits = await listVisits(patientId);
  const previousVisit = visits.find(v => v.id !== targetVisitId);
  if (!previousVisit) return;

  // Get prescription items from previous visit
  const snap = await getDocs(
    collection(db, 'patients', patientId, 'visits', previousVisit.id, 'prescriptionItems'),
  );

  // Add them to target visit
  const batch = [];
  for (const d of snap.docs) {
    const item = d.data();
    batch.push(
      addDoc(
        collection(db, 'patients', patientId, 'visits', targetVisitId, 'prescriptionItems'),
        { ...item, id: undefined, createdAt: serverTimestamp(), updatedAt: serverTimestamp() },
      ),
    );
  }
  await Promise.all(batch);
}

export async function saveSignature(
  patientId: string,
  visitId: string,
  dataUrl: string,
): Promise<string> {
  // Convert base64 data URL to blob
  const res = await fetch(dataUrl);
  const blob = await res.blob();
  const storageRef = ref(storage, `patients/${patientId}/visits/${visitId}/signature.png`);
  await uploadBytes(storageRef, blob);
  const url = await getDownloadURL(storageRef);
  await updateVisit(patientId, visitId, { signatureUrl: url });
  return url;
}

// Invoice number uses same atomic sequence pattern as patient ID
export async function generateInvoiceNumber(): Promise<string> {
  const seqRef = doc(db, 'sequences', 'invoices');
  let invoiceNumber = '';
  await runTransaction(db, async (tx) => {
    const snap = await tx.get(seqRef);
    const year = new Date().getFullYear();
    const storedYear = snap.exists() ? snap.data().year : year;
    const current = snap.exists() && storedYear === year ? snap.data().current : 0;
    const next = current + 1;
    tx.set(seqRef, { current: next, year }, { merge: true });
    invoiceNumber = `INV-${year}-${String(next).padStart(4, '0')}`;
  });
  return invoiceNumber;
}

**`src/lib/pdf/prescription.ts`:**

import jsPDF from 'jspdf';
import { format } from 'date-fns';

export async function generatePrescriptionPDF(params: {
  patient: Patient;
  visit: Visit;
  diagnoses: Diagnosis[];
  prescriptionItems: PrescriptionItem[];
  doctor: UserProfile;
  clinicSettings: Record<string, string>;
}): Promise<void> {
  const { patient, visit, diagnoses, prescriptionItems, doctor, clinicSettings } = params;
  const pdf = new jsPDF({ format: 'a4', unit: 'mm' });

  // Clinic header
  pdf.setFontSize(14).setFont('helvetica', 'bold');

## PART 13 â€” FIRESTORE SECURITY RULES

Create `firestore.rules` in project root:

```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuth() { return request.auth != null; }
    function role() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }
    function isAdmin()        { return isAuth() && role() == 'ADMIN'; }
    function isDoctor()       { return isAuth() && role() == 'DOCTOR'; }
    function isReceptionist() { return isAuth() && role() == 'RECEPTIONIST'; }
    function isClinical()     { return isAdmin() || isDoctor(); }

    // Users
    match /users/{uid} {
      allow read: if isAuth();
      allow write: if isAdmin();
      allow update: if request.auth.uid == uid;
    }

    // Patients â€” all roles can read demographics, only clinical can write
    match /patients/{patientId} {
      allow read: if isAuth();
      allow create, update: if isAuth();
      allow delete: if false; // soft delete only

      // Visits â€” receptionist cannot access
      match /visits/{visitId} {
        allow read, write: if isClinical();

        match /diagnoses/{id}     { allow read, write: if isClinical(); }
        match /vitals/{id}        { allow read, write: if isClinical(); }
        match /complaints/{id}    { allow read, write: if isClinical(); }
        match /prescriptionItems/{id} { allow read, write: if isClinical(); }
        match /documents/{id}     { allow read, write: if isAuth(); }
      }

      // Clinical history â€” clinical roles only
      match /clinicalHistory/{id} { allow read, write: if isClinical(); }
      match /familyHistory/{id}   { allow read, write: if isClinical(); }

      // Patient-level documents â€” all roles
      match /documents/{id} { allow read, write: if isAuth(); }
    }

    // Appointments â€” all roles
    match /appointments/{id} {
      allow read, write: if isAuth();
    }

    // Invoices â€” all roles
    match /invoices/{id} {
      allow read, write: if isAuth();
      match /payments/{id} { allow read, write: if isAuth(); }
    }

    // Medicines â€” all roles read, only clinical write
    match /medicines/{id} {
      allow read: if isAuth();
      allow write: if isClinical();
    }

    // Audit logs â€” admin read only, any write
    match /auditLogs/{id} {
      allow read: if isAdmin();
      allow create: if isAuth();
    }

    // Settings â€” admin write, all read
    match /clinicSettings/{key} {
      allow read: if isAuth();
      allow write: if isAdmin();
    }

    // Sequences â€” only transactional writes allowed
    match /sequences/{id} {
      allow read: if isAuth();
      allow write: if isAuth();
    }
  }
}
